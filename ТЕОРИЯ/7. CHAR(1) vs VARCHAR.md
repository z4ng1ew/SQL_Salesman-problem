## ⚠️ РАЗНИЦА ТИПОВ: CHAR(1) vs VARCHAR

```sql
-- В таблице:
CREATE TABLE tsp (
    point1 CHAR(1),      ← CHAR(1)
    point2 CHAR(1),      ← CHAR(1)
    ...
);

-- В массиве:
ARRAY['a']::varchar[]    ← VARCHAR[]

// Разные типы!
```

---

## 🔍 ПОЧЕМУ ЭТО РАБОТАЕТ?

### PostgreSQL АВТОМАТИЧЕСКИ преобразует типы!

```sql
-- PostgreSQL делает неявное приведение:
CHAR(1) → VARCHAR

-- Это безопасно, потому что:
CHAR(1) = 'a' (с возможным пробелом в конце)
VARCHAR = 'a' (без лишних пробелов)

-- При сравнении PostgreSQL "подгоняет" типы
```

---

## 📖 ОТЛИЧИЯ CHAR vs VARCHAR

### Таблица различий:

| Характеристика | CHAR(n) | VARCHAR(n) |
|----------------|---------|------------|
| **Длина** | Фиксированная (всегда n) | Переменная (до n) |
| **Заполнение** | Дополняет пробелами | Не дополняет |
| **Пустая строка** | Сохранит как n пробелов | Сохранит как '' |
| **Хранение** | Всегда n байт | 1-4 байта + длина |
| **Производительность** | Чуть быстрее для фикс. длины | Экономит место |

---

## 💡 ПРИМЕРЫ РАЗЛИЧИЙ

```sql
-- CHAR(5) - фиксированная длина
CREATE TABLE test_char (
    col CHAR(5)
);

INSERT INTO test_char VALUES ('ab');
SELECT col, LENGTH(col) FROM test_char;

-- Результат:
--  col   | length
-- -------+--------
--  ab    |      2  (PostgreSQL автоматически обрезает пробелы при выводе!)
-- 
-- Но в БД хранится: 'ab   ' (3 пробела в конце)


-- VARCHAR(5) - переменная длина
CREATE TABLE test_varchar (
    col VARCHAR(5)
);

INSERT INTO test_varchar VALUES ('ab');
SELECT col, LENGTH(col) FROM test_varchar;

-- Результат:
--  col | length
-- -----+--------
--  ab  |      2
--
-- В БД хранится: 'ab' (без пробелов)
```

## 🔬 ДЕТАЛЬНОЕ СРАВНЕНИЕ

```sql
-- ============================================
-- ЭКСПЕРИМЕНТ: CHAR vs VARCHAR
-- ============================================

-- CHAR(1)
SELECT 
    'a'::CHAR(1) AS значение,
    LENGTH('a'::CHAR(1)) AS длина,
    octet_length('a'::CHAR(1)) AS байт;
-- Результат:
--  значение | длина | байт
-- ----------+-------+------
--  a        |     1 |    1


-- VARCHAR
SELECT 
    'a'::VARCHAR AS значение,
    LENGTH('a'::VARCHAR) AS длина,
    octet_length('a'::VARCHAR) AS байт;
-- Результат:
--  значение | длина | байт
-- ----------+-------+------
--  a        |     1 |    1


-- CHAR(5) с коротким значением
SELECT 
    'ab'::CHAR(5) AS значение,
    LENGTH('ab'::CHAR(5)) AS длина,
    octet_length('ab'::CHAR(5)) AS байт,
    'ab'::CHAR(5) = 'ab     ' AS равны_с_пробелами;
-- Результат:
--  значение | длина | байт | равны_с_пробелами
-- ----------+-------+------+-------------------
--  ab       |     2 |    2 | true


-- VARCHAR(5) с коротким значением
SELECT 
    'ab'::VARCHAR(5) AS значение,
    LENGTH('ab'::VARCHAR(5)) AS длина,
    octet_length('ab'::VARCHAR(5)) AS байт;
-- Результат:
--  значение | длина | байт
-- ----------+-------+------
--  ab       |     2 |    2
```

## ❓ ЕСТЬ ЛИ КОНФЛИКТ ТИПОВ В НАШЕМ КОДЕ?

### Короткий ответ: ❌ НЕТ конфликта!

```sql
-- Таблица tsp:
point1 CHAR(1)  → 'a'

-- Массив:
ARRAY['a']::varchar[]  → {a}

-- При JOIN:
edges.point1 = routes.current_city
  ↓               ↓
CHAR(1)         VARCHAR

-- PostgreSQL автоматически приводит:
CHAR(1) → VARCHAR (неявное преобразование)
```

### Почему работает:

```sql
-- PostgreSQL делает так:
CHAR(1) 'a' → VARCHAR 'a'
           ↓
      сравнение
           ↓
      работает!
```

## 🎯 ПОЧЕМУ ИСПОЛЬЗУЕТСЯ VARCHAR[] В МАССИВЕ?

### 3 причины:

### 1️⃣ **Гибкость**
```sql
-- VARCHAR может хранить строки РАЗНОЙ длины
ARRAY['a', 'ab', 'abc']::varchar[]  ✅ работает

-- CHAR(1) только фиксированной
ARRAY['a', 'ab', 'abc']::char(1)[]  ❌ ошибка!
```

### 2️⃣ **Совместимость**
```sql
-- VARCHAR совместим с любыми текстовыми типами
CHAR(1) → VARCHAR  ✅ автоматически
TEXT → VARCHAR     ✅ автоматически
VARCHAR → CHAR(1)  ⚠️ нужна явная проверка длины
```

### 3️⃣ **Стандартная практика**
```sql
-- В PostgreSQL для массивов строк обычно используют varchar[]
-- Это общепринятая конвенция
```

## ✅ А МОЖНО ЛИ ИСПОЛЬЗОВАТЬ CHAR(1)[] ?

### ДА, можно! Вот доказательство:

```sql
-- Создаём файл для эксперимента
DROP TABLE IF EXISTS tsp CASCADE;

CREATE TABLE tsp (
    point1 CHAR(1),
    point2 CHAR(1),
    cost   INT
);

INSERT INTO tsp VALUES ('a','b',10), ('b','a',10);

-- ВАРИАНТ 1: С varchar[] (как сейчас)
WITH RECURSIVE test_varchar AS (
    SELECT 
        ARRAY['a']::varchar[] AS path,
        'a'::CHAR(1) AS current
    UNION ALL
    SELECT 
        path || ARRAY[t.point2]::varchar[],
        t.point2
    FROM test_varchar
    JOIN tsp t ON t.point1 = current
    WHERE NOT t.point2 = ANY(path)
    LIMIT 5
)
SELECT * FROM test_varchar;

-- ✅ РАБОТАЕТ!


-- ВАРИАНТ 2: С char(1)[] (альтернатива)
WITH RECURSIVE test_char AS (
    SELECT 
        ARRAY['a']::char(1)[] AS path,
        'a'::CHAR(1) AS current
    UNION ALL
    SELECT 
        path || ARRAY[t.point2]::char(1)[],
        t.point2
    FROM test_char
    JOIN tsp t ON t.point1 = current
    WHERE NOT t.point2 = ANY(path)
    LIMIT 5
)
SELECT * FROM test_char;

-- ✅ ТОЖЕ РАБОТАЕТ!
```

## 📊 СРАВНЕНИЕ: varchar[] vs char(1)[]

| Аспект | `varchar[]` | `char(1)[]` |
|--------|-------------|-------------|
| **Работает?** | ✅ Да | ✅ Да |
| **Гибкость** | ✅ Любая длина | ❌ Только 1 символ |
| **Совместимость** | ✅ Широкая | ⚠️ Узкая |
| **Стандарт** | ✅ Обычная практика | ⚠️ Редко используют |
| **Производительность** | Одинаковая | Одинаковая |

## 🔧 МОЖНО ЛИ УЛУЧШИТЬ НАШ КОД?

### Вариант 1: Оставить как есть (рекомендую)
```sql
-- Таблица: CHAR(1)
-- Массив: varchar[]
-- Причина: гибкость + совместимость
```

### Вариант 2: Использовать char(1)[]
```sql
-- Таблица: CHAR(1)
-- Массив: char(1)[]
-- Причина: строгая типизация

ARRAY['a']::char(1)[] AS visited_cities
```

### Вариант 3: Изменить таблицу на VARCHAR(1)
```sql
-- Таблица: VARCHAR(1)
-- Массив: varchar[]
-- Причина: единообразие типов

CREATE TABLE tsp (
    point1 VARCHAR(1),
    point2 VARCHAR(1),
    cost   INT
);
```

---

## 💡 МОЯ РЕКОМЕНДАЦИЯ

### ✅ Оставь как есть: CHAR(1) в таблице + varchar[] в массиве

**Почему:**
1. Работает без проблем (PostgreSQL приводит типы)
2. varchar[] более гибкий и универсальный
3. Стандартная практика в PostgreSQL
4. Нет влияния на производительность

**Когда менять на char(1)[]:**
- Если хочешь строгую типизацию
- Если уверен, что всегда будут 1-символьные города
- Для учебных целей (показать знание типов)

## 🧪 ФАЙЛ ДЛЯ ЭКСПЕРИМЕНТОВ

```sql
-- ==========================================
-- ЭКСПЕРИМЕНТЫ С ТИПАМИ CHAR vs VARCHAR
-- ==========================================

DROP TABLE IF EXISTS tsp CASCADE;

CREATE TABLE tsp (
    point1 CHAR(1),
    point2 CHAR(1),
    cost   INT
);

INSERT INTO tsp VALUES 
('a','b',10), ('b','a',10),
('a','c',15), ('c','a',15);

-- ============================================
-- ТЕСТ 1: VARCHAR[] (текущая версия)
-- ============================================
WITH RECURSIVE test1 AS (
    SELECT 
        ARRAY['a']::varchar[] AS path,
        'a'::CHAR(1) AS curr
    UNION ALL
    SELECT 
        path || ARRAY[t.point2]::varchar[],
        t.point2
    FROM test1
    JOIN tsp t ON t.point1 = curr
    WHERE NOT t.point2 = ANY(path) AND array_length(path, 1) < 3
)
SELECT 'varchar[]' AS тип, * FROM test1;

-- ============================================
-- ТЕСТ 2: CHAR(1)[] (альтернатива)
-- ============================================
WITH RECURSIVE test2 AS (
    SELECT 
        ARRAY['a']::char(1)[] AS path,
        'a'::CHAR(1) AS curr
    UNION ALL
    SELECT 
        path || ARRAY[t.point2]::char(1)[],
        t.point2
    FROM test2
    JOIN tsp t ON t.point1 = curr
    WHERE NOT t.point2 = ANY(path) AND array_length(path, 1) < 3
)
SELECT 'char(1)[]' AS тип, * FROM test2;

-- ============================================
-- ТЕСТ 3: Проверка совместимости типов
-- ============================================
SELECT 
    'Проверка JOIN между CHAR(1) и VARCHAR' AS описание;

SELECT 
    t.point1::text AS char_значение,
    pg_typeof(t.point1) AS char_тип,
    'a'::varchar AS varchar_значение,
    pg_typeof('a'::varchar) AS varchar_тип,
    t.point1 = 'a'::varchar AS совместимы
FROM tsp t
WHERE t.point1 = 'a'
LIMIT 1;

-- Результат:
--  char_значение | char_тип  | varchar_значение | varchar_тип | совместимы
-- ---------------+-----------+------------------+-------------+------------
--  a             | character | a                | text        | true

-- ✅ PostgreSQL приводит CHAR к VARCHAR автоматически!
```
